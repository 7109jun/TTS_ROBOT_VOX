cmake_minimum_required(VERSION 3.10)
project(robotvox C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
file(GLOB RV_SRC src/*.c)
add_executable(robotvox ${RV_SRC})
if(WIN32)
  target_link_libraries(robotvox shell32 winmm)
else()
  target_link_libraries(robotvox m ${CMAKE_DL_LIBS})
endif()
if(MSVC)
  target_compile_options(robotvox PRIVATE /W3 /utf-8)
else()
  target_compile_options(robotvox PRIVATE -Wall -Wextra -Wpedantic)
endif()
